import React, { useEffect, useState } from 'react'; import { supabase } from './supabase.js'; import Home from './Home.jsx'; import WineDetail from './WineDetail.jsx'; import EditWine from './EditWine.jsx'; import ScanBottle from './ScanBottle.jsx'; import AddManual from './AddManual.jsx'; import { uploadWinePhoto, wineFromDatabase } from './helpers.js'; import { cacheWine, cacheWines, queueChange, readCachedWines } from './offlineDb.js'; import { syncPendingChanges } from './offlineSync.js'; export default function App() { const [wines, setWines] = useState([]); const [screen, setScreen] = useState('home'); const [selectedWine, setSelectedWine] = useState(null); const [loading, setLoading] = useState(true); const [loadError, setLoadError] = useState(''); const [isOnline, setIsOnline] = useState(navigator.onLine); const [syncMessage, setSyncMessage] = useState(''); async function loadWines() { setLoading(true); setLoadError(''); const cached = await readCachedWines(); if (cached.length) { setWines(cached); setLoading(false); } if (!navigator.onLine) { if (!cached.length) setLoadError('No cached cellar data is available yet.'); return; } const { data, error } = await supabase .from('wines') .select('*') .order('wine_name', { ascending: true }); if (error) { if (!cached.length) setLoadError('Unable to connect to the cellar database.'); setLoading(false); return; } const freshWines = (data || []).map(wineFromDatabase); setWines(freshWines); await cacheWines(freshWines); setLoading(false); } useEffect(() => { loadWines(); async function handleOnline() { setIsOnline(true); setSyncMessage('Syncing offline changes…'); const result = await syncPendingChanges(); await loadWines(); setSyncMessage(result.synced ? `${result.synced} offline change${result.synced === 1 ? '' : 's'} synced.` : ''); setTimeout(() => setSyncMessage(''), 3000); } function handleOffline() { setIsOnline(false); setSyncMessage('Offline mode'); } window.addEventListener('online', handleOnline); window.addEventListener('offline', handleOffline); if (navigator.onLine) syncPendingChanges().then(() => loadWines()); return () => { window.removeEventListener('online', handleOnline); window.removeEventListener('offline', handleOffline); }; }, []); function openWine(wine) { setSelectedWine(wine); setScreen('detail'); } function updateWineInState(updatedWine) { setWines(current => current.map(wine => wine.id === updatedWine.id ? updatedWine : wine)); setSelectedWine(current => current?.id === updatedWine.id ? updatedWine : current); cacheWine(updatedWine); } async function changeQuantity(wine, delta) { const nextQuantity = Math.max(0, wine.quantity + delta); const optimisticWine = { ...wine, quantity: nextQuantity }; updateWineInState(optimisticWine); if (!navigator.onLine) { await queueChange({ operation: 'update', wineId: wine.id, payload: optimisticWine }); setSyncMessage('Change saved offline.'); return; } const { data, error } = await supabase .from('wines') .update({ quantity: nextQuantity, updated_at: new Date().toISOString() }) .eq('id', wine.id) .select('*') .single(); if (!error && data) { updateWineInState(wineFromDatabase(data)); return; } await queueChange({ operation: 'update', wineId: wine.id, payload: optimisticWine }); setSyncMessage('Change queued for sync.'); } async function savePhotoForWine(wine, file) { if (!navigator.onLine) { setSyncMessage('Photos require an internet connection.'); return wine; } const photoUrl = await uploadWinePhoto(file, wine.id); if (!photoUrl) return wine; const { data } = await supabase .from('wines') .update({ photo_url: photoUrl, updated_at: new Date().toISOString() }) .eq('id', wine.id) .select('*') .single(); const updatedWine = data ? wineFromDatabase(data) : { ...wine, photoUrl }; updateWineInState(updatedWine); return updatedWine; } async function createWine(details) { const wineDetails = details || {}; if (!navigator.onLine) { const temporaryWine = { id: `offline-${crypto.randomUUID()}`, producer: wineDetails.producer || '', name: wineDetails.wineName || wineDetails.wine_name || 'New wine', fullName: [wineDetails.producer, wineDetails.wineName || wineDetails.wine_name].filter(Boolean).join(' '), vintage: wineDetails.vintage || '', colour: wineDetails.colour || '', country: wineDetails.country || '', region: wineDetails.region || '', appellation: wineDetails.appellation || '', size: wineDetails.bottleSize || wineDetails.bottle_size || '750ml', quantity: Number(wineDetails.quantity || 1), drinkFrom: wineDetails.drinkFrom || '', drinkTo: wineDetails.drinkTo || '', notes: wineDetails.notes || '', photoUrl: '' }; await cacheWine(temporaryWine); await queueChange({ operation: 'create', wineId: temporaryWine.id, payload: wineDetails }); setWines(current => [temporaryWine, ...current]); openWine(temporaryWine); setSyncMessage('Wine saved offline and queued for sync.'); return temporaryWine; } const { data, error } = await supabase .from('wines') .insert({ producer: wineDetails.producer || '', wine_name: wineDetails.wineName || wineDetails.wine_name || 'New wine', vintage: wineDetails.vintage || '', colour: wineDetails.colour || '', country: wineDetails.country || '', region: wineDetails.region || '', appellation: wineDetails.appellation || '', bottle_size: wineDetails.bottleSize || wineDetails.bottle_size || '750ml', quantity: Number(wineDetails.quantity || 1), drinking_from: wineDetails.drinkFrom ? Number(wineDetails.drinkFrom) : null, drinking_to: wineDetails.drinkTo ? Number(wineDetails.drinkTo) : null, notes: [wineDetails.notes || '', wineDetails.subregion ? `Subregion: ${wineDetails.subregion}` : '', wineDetails.storageLocation ? `Storage location: ${wineDetails.storageLocation}` : ''].filter(Boolean).join('\n') }) .select('*') .single(); if (error || !data) { console.error('Create wine error:', error); return null; } let newWine = wineFromDatabase(data); if (wineDetails.photoFile) { newWine = await savePhotoForWine(newWine, wineDetails.photoFile); } await cacheWine(newWine); setWines(current => [newWine, ...current.filter(wine => wine.id !== newWine.id)]); openWine(newWine); return newWine; } async function saveEditedWine(wine, details) { const optimisticWine = { ...wine, producer: details.producer || '', name: details.wineName || 'New wine', fullName: [details.producer, details.wineName].filter(Boolean).join(' '), vintage: details.vintage || '', colour: details.colour || '', country: details.country || '', region: details.region || '', appellation: details.appellation || '', size: details.bottleSize || '750ml', quantity: Number(details.quantity || 0), drinkFrom: details.drinkFrom || '', drinkTo: details.drinkTo || '', notes: details.notes || '' }; updateWineInState(optimisticWine); if (!navigator.onLine || wine.id.startsWith('offline-')) { await queueChange({ operation: wine.id.startsWith('offline-') ? 'create' : 'update', wineId: wine.id, payload: optimisticWine }); openWine(optimisticWine); setSyncMessage('Changes saved offline.'); return optimisticWine; } const { data, error } = await supabase .from('wines') .update({ producer: details.producer || '', wine_name: details.wineName || 'New wine', vintage: details.vintage || '', colour: details.colour || '', country: details.country || '', region: details.region || '', appellation: details.appellation || '', bottle_size: details.bottleSize || '750ml', quantity: Number(details.quantity || 0), drinking_from: details.drinkFrom ? Number(details.drinkFrom) : null, drinking_to: details.drinkTo ? Number(details.drinkTo) : null, notes: details.notes || '', updated_at: new Date().toISOString() }) .eq('id', wine.id) .select('*') .single(); if (error || !data) { await queueChange({ operation: 'update', wineId: wine.id, payload: optimisticWine }); setSyncMessage('Changes queued for sync.'); openWine(optimisticWine); return optimisticWine; } const updatedWine = wineFromDatabase(data); updateWineInState(updatedWine); openWine(updatedWine); return updatedWine; } if (screen === 'detail' && selectedWine) { return ( setScreen('home')} onAddOne={() => changeQuantity(selectedWine, 1)} onConsumeOne={() => changeQuantity(selectedWine, -1)} onEdit={() => setScreen('edit')} savePhotoForWine={savePhotoForWine} /> ); } if (screen === 'edit' && selectedWine) { return ( setScreen('detail')} onSave={saveEditedWine} /> ); } if (screen === 'scan') { return ( setScreen('home')} onOpenWine={openWine} onCreateWine={createWine} savePhotoForWine={savePhotoForWine} /> ); } if (screen === 'manual') { return ( setScreen('home')} onCreateWine={createWine} /> ); } return ( setScreen('scan')} onAddManual={() => setScreen('manual')} isOnline={isOnline} syncMessage={syncMessage} /> ); }
